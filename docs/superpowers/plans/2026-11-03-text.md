**Goal:** Build the offline-ready Text competition suite (`utils/text/`) on top of `utils/core/`, providing `TextConfig`, stateless Indonesian/general text cleaning and row-wise/pair feature engineering, `TfidfPipeline` (sparse n-grams + linear/LGBM estimators), `TransformerPipeline` (native PyTorch AMP fine-tuning with fresh per-fold weights and two-tier time budget guard), OOF error analysis, and multi-label execution.

**Architecture:** `TextConfig(BaseConfig)` serializes text-specific parameters under `extra["_text"]` so Core's `BaseConfig.load` (`load_runs`/`blend`) and `RunManager(resume_from=...)` work seamlessly while preserving user keys in `extra`. `TextPipelineBase(BasePipeline)` provides shared input/binary-target validation, `estimate_fold_seconds` pre-fold time budgeting, stateless `build_features`, per-fold numeric `StandardScaler` in `fit_transform_fold`, and `NaN`-safe binary threshold tuning in `postprocess`. `TfidfPipeline` keeps word/char n-gram matrices and leak-free pair TF-IDF cosine similarities as `scipy.sparse.csr_matrix` inside `fit_predict_fold`, while `TransformerPipeline` loads a fresh pretrained HuggingFace model from disk per fold (`build_model(fold)`) and trains via a native PyTorch loop using Core's `self._run_start` timer and `cfg.select_metric`.

**Tech Stack:** Python 3.10+, `numpy`, `pandas`, `scipy.sparse`, `scikit-learn`, `lightgbm`, `torch`, `transformers`.

```python
def test_core_pre_fold_estimate_per_fold_seed_and_truncated_report(tmp_path):
    tr, te = make_bin_df(45)
    clock = FakeClock(start=10.0)
    recorded_starts = []
    fold_random_Draws = {}

    class GuardedPipe(SklearnBinaryPipe):
        def estimate_fold_seconds(self, completed_folds):
            return max(self.fold_info[fk].get("duration_sec", 0.0) for fk in completed_folds) * 1.1

        def build_model(self, fold):
            recorded_starts.append(self._run_start)
            fold_random_Draws[fold] = float(np.random.rand())
            return super().build_model(fold)

        def fit_predict_fold(self, model, tr_df, va_df, te_df, fold):
            clock.advance(40.0)
            if fold == 0:
                self.fold_info[fold]["truncated"] = True
            return super().fit_predict_fold(model, tr_df, va_df, te_df, fold)

    # max_minutes=1.2 (72s): fold 0 takes 40s -> rem_sec=32s < est_sec (40s * 1.1 = 44s) -> stops before fold 1!
    cfg_part = BaseConfig(
        run_name="guard_part", task="binary", target="y", metric="auc",
        n_folds=3, max_minutes=1.2, runs_dir=str(tmp_path),
    )
    pipe_part = GuardedPipe(cfg_part)
    pipe_part._clock = clock
    r_part = pipe_part.run(tr, te)

    assert recorded_starts == [10.0]
    assert r_part.scores["status"] == "partial"
    assert r_part.scores["n_folds_done"] == 1
    assert r_part.scores["per_fold"]["0"].get("truncated") is True
    assert "0 (truncated)" in (r_part.run_dir / "summary.md").read_text(encoding="utf-8")

    # Resume and verify fold 1 RNG draw matches an uninterrupted full run's fold 1 RNG draw
    cfg_res = BaseConfig(
        run_name="guard_part", task="binary", target="y", metric="auc",
        n_folds=3, max_minutes=10.0, resume_from=str(r_part.run_dir), runs_dir=str(tmp_path),
    )
    pipe_res = GuardedPipe(cfg_res)
    pipe_res._clock = clock
    pipe_res.run(tr, te)
    draw_fold1_resumed = fold_random_Draws[1]

    fold_random_Draws.clear()
    cfg_full = BaseConfig(
        run_name="guard_full", task="binary", target="y", metric="auc",
        n_folds=3, max_minutes=10.0, runs_dir=str(tmp_path),
    )
    pipe_full = GuardedPipe(cfg_full)
    pipe_full._clock = clock
    pipe_full.run(tr, te)
    assert draw_fold1_resumed == fold_random_Draws[1]
```

```python
# tests/text/helpers.py
from pathlib import Path
import numpy as np
import pandas as pd


def make_text_bin_df(n: int = 48) -> tuple[pd.DataFrame, pd.DataFrame]:
    rng = np.random.default_rng(42)
    pos_Pool = ["produk sangat bagus mantap cepat puas sekali", "layanan ramah barang asli berkualitas tinggi"]
    neg_Pool = ["barang rusak parah kecewa berat pengiriman lambat", "respon buruk sekali penipuan zonk"]
    y = np.array([i % 2 for i in range(n)], dtype=int)
    texts = [
        f"{pos_Pool[i % 2] if y[i] == 1 else neg_Pool[i % 2]} #{i}"
        for i in range(n)
    ]
    tr = pd.DataFrame({"id": range(n), "text": texts, "num_meta": rng.normal(size=n), "cat_meta": ["A" if i % 3 == 0 else "B" for i in range(n)], "y": y})
    te_texts = [f"{pos_Pool[i % 2]} test {i}" if i % 2 == 1 else f"{neg_Pool[i % 2]} test {i}" for i in range(12)]
    te = pd.DataFrame({"id": range(100, 112), "text": te_texts, "num_meta": rng.normal(size=12), "cat_meta": ["A" if i % 2 == 0 else "C" for i in range(12)]})
    return tr, te


def create_tiny_hf_model_dir(base_dir: Path) -> Path:
    """Create a tiny 100% offline BERT model and WordPiece tokenizer on disk for fast CPU unit tests."""
    from transformers import BertConfig, BertForSequenceClassification, BertTokenizerFast

    model_dir = base_dir / "tiny_bert"
    model_dir.mkdir(parents=True, exist_ok=True)

    vocab_tokens = [
        "[PAD]", "[UNK]", "[CLS]", "[SEP]", "[MASK]",
        "produk", "sangat", "bagus", "mantap", "cepat", "puas", "sekali",
        "layanan", "ramah", "barang", "asli", "berkualitas", "tinggi",
        "rusak", "parah", "kecewa", "berat", "pengiriman", "lambat",
        "respon", "buruk", "penipuan", "zonk", "test", "tidak", "yang",
        "col", "val", "a", "b", "c", "0", "1", "2",
    ]
    while len(vocab_tokens) < 64:
        vocab_tokens.append(f"tok{len(vocab_tokens)}")
    (model_dir / "vocab.txt").write_text("\n".join(vocab_tokens) + "\n", encoding="utf-8")

    tok = BertTokenizerFast(vocab_file=str(model_dir / "vocab.txt"), do_lower_case=True)
    tok.save_pretrained(str(model_dir))

    cfg = BertConfig(
        vocab_size=len(vocab_tokens),
        hidden_size=16,
        num_hidden_layers=1,
        num_attention_heads=2,
        intermediate_size=32,
        max_position_embeddings=128,
        pad_token_id=0,
    )
    model = BertForSequenceClassification(cfg)
    model.save_pretrained(str(model_dir))
    return model_dir
```

```python
# tests/text/test_analysis_and_multilabel.py
import pandas as pd
import pytest

from tests.text.helpers import create_tiny_hf_model_dir, make_text_bin_df
from utils.core.ensemble import blend, load_runs
from utils.text.analysis import extract_top_errors, run_multilabel
from utils.text.config import TextConfig
from utils.text.tfidf import TfidfPipeline
from utils.text.transformer import TransformerPipeline


def test_extract_top_errors_creates_csv_and_sorts_descending(tmp_path):
    tr, te = make_text_bin_df(36)
    cfg = TextConfig(task="binary", target="y", metric="auc", estimator="ridge", n_folds=3, runs_dir=str(tmp_path))
    res = TfidfPipeline(cfg).run(tr, te)

    err_df = extract_top_errors(res, tr, n=10)
    assert len(err_df) == 10
    assert (res.run_dir / "error_analysis.csv").exists()
    assert "text" in err_df.columns and "error" in err_df.columns
    assert list(err_df["error"]) == sorted(err_df["error"], reverse=True)


def test_run_multilabel_two_labels_two_pipelines_and_blend(tmp_path):
    model_dir = create_tiny_hf_model_dir(tmp_path)
    tr, te = make_text_bin_df(36)
    tr["label_a"] = tr["y"]
    tr["label_b"] = 1 - tr["y"]
    folds_dir = tmp_path / "shared_folds"
    runs_dir = tmp_path / "runs"

    sample_sub = pd.DataFrame({"id": list(reversed(te["id"].tolist())), "label_a": 0.0, "label_b": 0.0})

    # 1. Run TfidfPipeline across both labels (creates folds/label_a/folds.csv and folds/label_b/folds.csv)
    cfg_tfidf = TextConfig(
        run_name="tfidf",
        task="binary",
        target="label_a",
        metric="auc",
        estimator="logistic",
        n_folds=2,
        runs_dir=str(runs_dir),
    )
    res_tfidf = run_multilabel(
        TfidfPipeline,
        cfg_tfidf,
        tr,
        te,
        target_cols=["label_a", "label_b"],
        folds_dir=folds_dir,
        sample_sub=sample_sub,
        sub_path=tmp_path / "sub_tfidf.csv",
    )
    assert (folds_dir / "label_a" / "folds.csv").exists()
    assert (folds_dir / "label_b" / "folds.csv").exists()
    sub_df = pd.read_csv(tmp_path / "sub_tfidf.csv")
    assert list(sub_df["id"]) == list(sample_sub["id"])
    assert list(sub_df.columns) == ["id", "label_a", "label_b"]

    # 2. Run TransformerPipeline across the same 2 labels reusing folds_dir -> folds_hash matches per label!
    cfg_hf = TextConfig(
        run_name="hf",
        task="binary",
        target="label_a",
        metric="auc",
        model_name_or_path=str(model_dir),
        max_length=24,
        epochs=1,
        train_batch_size=8,
        n_folds=2,
        runs_dir=str(runs_dir),
    )
    res_hf = run_multilabel(
        TransformerPipeline,
        cfg_hf,
        tr,
        te,
        target_cols=["label_a", "label_b"],
        folds_dir=folds_dir,
    )

    # 3. Blend TfidfPipeline + TransformerPipeline for each label without FoldMismatchError
    for lbl in ["label_a", "label_b"]:
        ens = load_runs([res_tfidf[lbl].run_dir, res_hf[lbl].run_dir])
        blended = blend(ens, method="weights", runs_dir=runs_dir, run_name=f"blend_{lbl}")
        assert blended.scores["status"] == "ok"
```
